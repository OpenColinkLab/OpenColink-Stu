void TA1_N_IRQHandler (void){
if(int_cap_status == TAxCCR4_CCIFG)
    {
        uint16_t index = 0;
        #ifdef DBG_MOD_OFF
            SOL_Timer_A_clearTAxR(TIMER_A1_BASE);
            cap_val3 = MAP_Timer_A_getCaptureCompareCount(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
            USER_DBG("cap_val3 = %d[MAP_Timer_A_getCaptureCompareCount]\n",cap_val3);
            cap_val3 = MAP_Timer_A_getCounterValue(TIMER_A1_BASE);
            USER_DBG("cap_val3 = %d[MAP_Timer_A_getCaptureCompareCount]\n",cap_val3);

            SOL_Timer_A_clearCCRx(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
            cap_val3 = MAP_Timer_A_getCounterValue(TIMER_A1_BASE);
            USER_DBG("cap_val3 = %d[Timer_A_getCounterValue]\n",cap_val3);
            cap_val3 = MAP_Timer_A_getCaptureCompareCount(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
            USER_DBG("cap_val3 = %d[Timer_A_getCounterValue]\n",cap_val3);
            for ( index = 0; index < 20; index++)
            {
                cap_val3 = MAP_Timer_A_getCaptureCompareCount(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
                USER_DBG("cap_val3 = %d[Timer_A_getCounterValue][%d]\n",cap_val3,index);
            }
            
        #endif
        if(int_cap_status == TAxCCR4_CCIFG)
        {
            switch (freq_cap_status)
            {
                case CAP_START:
                        freq_cap_status = CAP_VAL1;
                        #ifdef DBG_MOD_OFF
                            USER_DBG("counter clear\n");
                        #endif
                    break;

                case CAP_VAL1:
                        cap_val1 = MAP_Timer_A_getCaptureCompareCount(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
                        freq_cap_status = CAP_VAL2;
                        #ifdef DBG_MOD_OFF
                            USER_DBG("cap_val1 = %d\n",cap_val1);
                        #endif
                    break;

                case CAP_VAL2:
                        cap_val2 = MAP_Timer_A_getCaptureCompareCount(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
                        freq_cap_status = CAP_END;
                        #ifdef DBG_MOD_OFF
                            USER_DBG("cap_val2 = %d\n",cap_val2);
                        #endif
                    break;

                case CAP_END:
                        #ifdef DBG_MOD
                            USER_DBG("end of get cap_val\n");
                            cap_val3 = MAP_Timer_A_getCaptureCompareCount(TIMER_A1_BASE,TIMER_A_CAPTURECOMPARE_REGISTER_4);
                            USER_DBG("cap_val3 = %d\n",cap_val3);
                            printf("Freq = %f\ncap_val1 = %d\ncap_val2 = %d\n",48000000 / (float)(cap_val2 - cap_val1),cap_val1,cap_val2);

                            printf("\n\n**************************************\n");
                        #endif
                        freq_cap_status = CAP_START;
                        //关闭TA0中断,在START中打开
                        //MAP_Interrupt_disableInterrupt(INT_TA1_N);
                    break;

                default:
                    break;
            }
        }
    }
}
