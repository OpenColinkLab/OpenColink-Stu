/*-------------------------*\
      TA0_0_IRQHandler(2)
\*-------------------------*/
void TA0_0_IRQHandler (void){
/*---flag clear---*/
    uint32_t int_cap_flag;
    uint32_t int_ove_flag;
    //获取中断标志
    int_ove_flag = TA0CTL & 0x01;   //溢出标志 (&)
    int_cap_flag = TA0CCTL4 & 0x01; //捕获标志 (&)

    //清除溢出标志位
    TA0CTL &= ~ 0x01;
    //清除捕获比较中断标志位
    TA0CCTL4 &= ~ 0x01;
//-------------------------------------------------------//
    

#ifdef DBG_MOD_OFF
    //USER_DBG("In TA0_0_IRQHandler"); //在这里串口打印非常影响延时特别是放在刚进中断的时候
#endif
}
